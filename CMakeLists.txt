cmake_minimum_required(VERSION 3.18)
project(QLEAF LANGUAGES CXX)

include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "15.2.0")
  message(WARNING "Q-LEAF was tested with GCC 15.2.0.")
endif()
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)

function(qleaf_target target)
  target_compile_options(${target} PRIVATE -O3 -march=native -Wall -Wextra)
  target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX Threads::Threads)
endfunction()

add_executable(qleaf_bench src/qleaf_bench.cpp)
target_include_directories(qleaf_bench PRIVATE src)
qleaf_target(qleaf_bench)

option(QLEAF_BUILD_TOOLS "Build graph and entry-point generators" ON)
if(QLEAF_BUILD_TOOLS)
  set(FAISS_ROOT "" CACHE PATH "FAISS 1.12.0 source/build tree or installation prefix")
  set(QLEAF_OPENBLAS "" CACHE FILEPATH "OpenBLAS library")
  set(QLEAF_GFORTRAN "" CACHE FILEPATH "Fortran runtime for OpenBLAS")
  find_path(FAISS_INCLUDE_DIR faiss/Index.h HINTS "${FAISS_ROOT}" "${FAISS_ROOT}/include" REQUIRED)
  find_library(FAISS_LIBRARY NAMES faiss HINTS "${FAISS_ROOT}/build/faiss" "${FAISS_ROOT}/lib" "${FAISS_ROOT}/lib64" REQUIRED)
  file(STRINGS "${FAISS_INCLUDE_DIR}/faiss/Index.h" _faiss_version REGEX "^#define FAISS_VERSION_(MAJOR|MINOR|PATCH) ")
  if(NOT _faiss_version MATCHES "FAISS_VERSION_MAJOR 1(;|$)" OR
     NOT _faiss_version MATCHES "FAISS_VERSION_MINOR 12(;|$)" OR
     NOT _faiss_version MATCHES "FAISS_VERSION_PATCH 0(;|$)")
    message(FATAL_ERROR "FAISS 1.12.0 is required; use commit 2135e5a28d0f2a932dd02b64b566ac1a52266540.")
  endif()
  if(NOT QLEAF_OPENBLAS)
    find_library(_qleaf_openblas NAMES openblas REQUIRED)
    set(QLEAF_OPENBLAS "${_qleaf_openblas}")
  endif()
  if(NOT QLEAF_GFORTRAN)
    find_library(_qleaf_gfortran NAMES gfortran libgfortran.so.5 REQUIRED)
    set(QLEAF_GFORTRAN "${_qleaf_gfortran}")
  endif()
  find_package(Boost REQUIRED)
  set(_nsg "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nsg")
  set(_efanna "${CMAKE_CURRENT_SOURCE_DIR}/third_party/efanna_graph")

  add_library(qleaf_nsg STATIC "${_nsg}/src/index.cpp" "${_nsg}/src/index_nsg.cpp")
  add_library(qleaf_efanna STATIC "${_efanna}/src/index.cpp" "${_efanna}/src/index_graph.cpp" "${_efanna}/src/index_kdtree.cpp" "${_efanna}/src/index_random.cpp")
  foreach(target qleaf_nsg qleaf_efanna)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 11)
    target_compile_definitions(${target} PRIVATE INFO)
    target_compile_options(${target} PRIVATE -O3 -march=native -Wall)
    target_include_directories(${target} SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
    target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX Threads::Threads)
  endforeach()
  target_include_directories(qleaf_nsg PRIVATE "${_nsg}/include")
  target_include_directories(qleaf_efanna PRIVATE "${_efanna}/include")
  target_compile_options(qleaf_nsg PRIVATE -fno-builtin-malloc -fno-builtin-calloc -fno-builtin-realloc -fno-builtin-free)

  add_library(qleaf_loader STATIC src/compat/nsg_qepo_data_loader.cpp)
  add_library(qleaf_utils STATIC src/compat/nsg_qepo_utils.cpp)
  foreach(target qleaf_loader qleaf_utils)
    target_include_directories(${target} PRIVATE src/compat "${_nsg}/include" "${FAISS_INCLUDE_DIR}")
    target_include_directories(${target} SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
    qleaf_target(${target})
  endforeach()

  foreach(target compute_ep_nsg build_nsg_subgraphs build_nsg_subgraphs_ml)
    add_executable(${target} "src/${target}.cpp")
    if(target STREQUAL "compute_ep_nsg")
      target_include_directories(${target} PRIVATE src/compat "${_nsg}/include" "${FAISS_INCLUDE_DIR}")
      target_link_libraries(${target} PRIVATE qleaf_loader qleaf_utils qleaf_nsg)
    elseif(target STREQUAL "build_nsg_subgraphs")
      target_include_directories(${target} PRIVATE "${_efanna}/include" "${_nsg}/include" src/compat "${FAISS_INCLUDE_DIR}")
      target_link_libraries(${target} PRIVATE qleaf_loader qleaf_efanna qleaf_nsg
        "${QLEAF_OPENBLAS}" "${QLEAF_GFORTRAN}" qleaf_utils "${FAISS_LIBRARY}")
    else()
      # Both libraries use efanna2e; the builder requires NN-Descent headers first.
      target_include_directories(${target} PRIVATE "${_efanna}/include" "${_nsg}/include" src/compat "${FAISS_INCLUDE_DIR}")
      target_link_libraries(${target} PRIVATE qleaf_loader qleaf_utils qleaf_efanna qleaf_nsg)
    endif()
    target_include_directories(${target} SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
    if(NOT target STREQUAL "build_nsg_subgraphs")
      target_link_libraries(${target} PRIVATE "${FAISS_LIBRARY}" "${QLEAF_OPENBLAS}" "${QLEAF_GFORTRAN}")
    endif()
    qleaf_target(${target})
  endforeach()

  # Retain the effective options of the original construction libraries.
  foreach(target qleaf_nsg qleaf_loader qleaf_utils build_nsg_subgraphs)
    target_compile_options(${target} PRIVATE
      -fvisibility-inlines-hidden -fmessage-length=0 -mtune=haswell
      -ftree-vectorize -fPIC -fstack-protector-strong -fno-plt
      -ffunction-sections -pipe)
  endforeach()
  foreach(target qleaf_loader qleaf_utils build_nsg_subgraphs)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS ON)
    target_compile_options(${target} PRIVATE -UNDEBUG -g)
  endforeach()
  target_compile_options(qleaf_efanna PRIVATE -UNDEBUG)
  target_link_options(build_nsg_subgraphs PRIVATE
    -Wl,-O2 -Wl,--sort-common -Wl,--as-needed -Wl,-z,relro -Wl,-z,now
    -Wl,--disable-new-dtags -Wl,--gc-sections -Wl,--allow-shlib-undefined)
endif()
